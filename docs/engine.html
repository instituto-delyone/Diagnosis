// Variáveis de Estado let pontuacao = 0; let estabilidade = 85; let turnos = 0; let casoAtual = null; // Elementos da UI const clinicalLog = document.getElementById('clinical-log'); const userInput = document.getElementById('user-input'); const healthBar = document.getElementById('health-bar'); const healthDisplay = document.getElementById('health-display'); const scoreDisplay = document.getElementById('score-display'); const valFc = document.getElementById('val-fc'); const valSpo2 = document.getElementById('val-spo2'); const valPa = document.getElementById('val-pa'); // 1. Catálogo de Arquivos JSON (Mapeie seus arquivos do GitHub aqui) const arquivosPorSala = { vermelha: ['pcr.json', 'sepse_e_choque_septico.json', 'cardiologia.json', 'toxicologia.json'], clinica: ['nefrologia.json', 'dispneia.json', 'endocrinologia.json', 'reumatologia.json', 'neurologia.json'] }; // 2. Descobrir em qual sala estamos pela URL (ex: engine.html?sala=vermelha) const urlParams = new URLSearchParams(window.location.search); const salaEscolhida = urlParams.get('sala') || 'clinica'; // Padrão é clínica se não houver parâmetro window.onload = () => { carregarNovoCaso(); // Carrega o primeiro caso ao abrir a página }; // 3. Função principal que sorteia o arquivo, faz o fetch e inicia o caso async function carregarNovoCaso() { clinicalLog.innerHTML = ''; adicionarMensagem("preceptor", "🔄 Conectando ao repositório de casos..."); try { // Sorteia um arquivo da lista da sala escolhida const listaArquivos = arquivosPorSala[salaEscolhida]; const arquivoSorteado = listaArquivos[Math.floor(Math.random() * listaArquivos.length)]; // Faz o Fetch do JSON na pasta knowledge_base // Certifique-se de que o caminho 'knowledge_base/' está correto no seu GitHub const response = await fetch(`knowledge_base/${arquivoSorteado}`); const casosDoArquivo = await response.json(); // Sorteia um caso específico dentro do arquivo JSON const casoSorteado = casosDoArquivo[Math.floor(Math.random() * casosDoArquivo.length)]; iniciarCaso(casoSorteado); } catch (error) { adicionarMensagem("preceptor", "❌ Erro ao carregar os casos. Verifique a conexão ou os nomes dos arquivos JSON."); console.error(error); } } function iniciarCaso(caso) { casoAtual = caso; estabilidade = 85; pontuacao = 0; turnos = 0; clinicalLog.innerHTML = ''; // Adaptado para ler a "vinheta_admissao" dos seus JSONs reais adicionarMensagem("preceptor", `Iniciando caso de Dificuldade: ${casoAtual.dificuldade}. Qual sua conduta inicial?`); adicionarMensagem("paciente", `Admissão: ${casoAtual.vinheta_admissao}`); atualizarDashboard(); } // A função Pular Caso reaproveita a lógica do botão Próximo function proximoCaso() { carregarNovoCaso(); } // Função para interagir com o engine.js e processar a resposta do usuário function processarAcao() { const inputAluno = userInput.value.trim(); if (!inputAluno || !casoAtual) return; turnos++; adicionarMensagem("user", "Conduta: " + inputAluno); let acaoReconhecida = false; // Aqui você pode expandir a lógica para ler 'fase_1_investigacao', 'fase_2_diagnostico', etc. // Simulando a validação baseada no modelo atual: const checkInvestigacao = idmtEngine.validarResposta(inputAluno, casoAtual.fase_1_investigacao.gabarito_esperado); if (checkInvestigacao.valido) { adicionarMensagem("preceptor", casoAtual.fase_1_investigacao.achado_sucesso); estabilidade = Math.min(100, estabilidade + 10); pontuacao += 25; acaoReconhecida = true; } if (!acaoReconhecida) { adicionarMensagem("preceptor", casoAtual.fase_1_investigacao.resposta_preceptor_erro || "Conduta não prioritária agora."); estabilidade -= 10; } atualizarDashboard(); userInput.value = ''; clinicalLog.scrollTop = clinicalLog.scrollHeight; } function atualizarDashboard() { if (estabilidade <= 0) estabilidade = 0; healthDisplay.innerText = estabilidade; healthBar.style.width = estabilidade + '%'; scoreDisplay.innerText = Math.max(0, Math.floor(pontuacao)); // Atualização dos sinais vitais conforme a estabilidade if (estabilidade < 50) { valFc.innerText = "140"; valSpo2.innerText = "85"; valPa.innerText = "70/40"; healthBar.style.backgroundColor = '#dc2626'; } else { valFc.innerText = "110"; valSpo2.innerText = "93"; valPa.innerText = "98/60"; healthBar.style.backgroundColor = estabilidade > 60 ? 'var(--idmt-green)' : '#f59e0b'; } } function pedirAjuda() { if(!casoAtual) return; pontuacao = Math.max(0, pontuacao - 10); adicionarMensagem("preceptor", `Dica: ${casoAtual.fase_1_investigacao.resposta_preceptor_erro}`); atualizarDashboard(); } function adicionarMensagem(classe, texto) { const div = document.createElement('div'); div.className = 'mensagem ' + classe; div.innerText = texto; clinicalLog.appendChild(div); }
